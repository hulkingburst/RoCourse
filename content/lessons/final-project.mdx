```
CLIENT                          SERVER
your action ──FireServer──▶    Handler: verify, decide, apply
UI labels  ◀──FireClient───    every change pushes updated totals
                               SaveSystem: load on join, save on leave
```

**GameConfig** is the single source of truth for every number in both games:

```luau
local config = {
    -- Coin Tycoon
    coinsPerClick = 1,
    upgradeBaseCost = 25,

    -- The Collector
    coinsPerPickup = 1,
    petBonusPerLevel = 1,
    petBaseCost = 30,

    -- shared
    coinsKey = "Coins",
    levelKey = "Level",
}

return config
```

In Coin Tycoon, `Level` is your click multiplier. In The Collector, `Level` is how many pets you own. Change one number here and the whole economy reprices.

## Step 1 — the math module

Both games run on the same module — it just serves two economies. All the money math in one testable place:

```luau
local GameMath = {}

-- Coin Tycoon: what one click is worth
function GameMath.rewardForClick(multiplier)
    return 1 * multiplier
end

-- The Collector: what one pickup is worth
function GameMath.coinsPerPickup(petLevel)
    return 1 + petLevel
end

-- shared: how prices climb
function GameMath.upgradeCost(baseCost, currentLevel)
    return baseCost * (currentLevel + 1)
end

return GameMath
```

Every script that needs a price or a reward calls these functions — no duplicated formulas. If the economy feels wrong, you change *one* file.

## Step 2 — SaveSystem

The persistence layer, using your datastores knowledge. It saves *two* values as one table — and it's identical for both games, because both games track `Coins` and `Level`:

```luau
local Players = game:GetService("Players")
local DataStoreService = game:GetService("DataStoreService")
local Shared = game:GetService("ReplicatedStorage").Shared
local GameConfig = require(Shared.GameConfig)

local saveStore = DataStoreService:GetDataStore("FinalProjectSaves")

function createStats(player, coins, level)
    local stats = Instance.new("Folder")
    stats.Name = "leaderstats"
    stats.Parent = player

    local coinsValue = Instance.new("NumberValue")
    coinsValue.Name = GameConfig.coinsKey
    coinsValue.Value = coins
    coinsValue.Parent = stats

    local levelValue = Instance.new("NumberValue")
    levelValue.Name = GameConfig.levelKey
    levelValue.Value = level
    levelValue.Parent = stats
end

Players.PlayerAdded:Connect(function(player)
    local data = saveStore:GetAsync(player.UserId)
    if data == nil then
        data = { coins = 0, level = 1 }
    end
    createStats(player, data.coins, data.level)
end)

Players.PlayerRemoving:Connect(function(player)
    local coins = player.leaderstats[GameConfig.coinsKey].Value
    local level = player.leaderstats[GameConfig.levelKey].Value
    saveStore:SetAsync(player.UserId, { coins = coins, level = level })
end)
```

Load on join, save on leave, default for new players — the datastores lesson, upgraded to a table.

## Step 3 — Economy

Clicks or pickups and purchases, all through the server. Each handler is tiny: **verify, decide, apply, report.** All the noise lives in the modules.

<Variant name="tycoon">

```luau
local ReplicatedStorage = game:GetService("ReplicatedStorage")
local Shared = ReplicatedStorage.Shared
local GameConfig = require(Shared.GameConfig)
local GameMath = require(Shared.GameMath)

local clickEvent = ReplicatedStorage.RemoteEvents.ClickCoin
local buyEvent = ReplicatedStorage.RemoteEvents.BuyUpgrade

function getStats(player)
    local stats = player.leaderstats
    if not stats then return nil, nil end
    return stats[GameConfig.coinsKey], stats[GameConfig.levelKey]
end

function pushUpdate(player)
    local coins, level = getStats(player)
    if coins and level then
        clickEvent:FireClient(player, coins.Value, level.Value)
    end
end

clickEvent.OnServerEvent:Connect(function(player)
    local coins, level = getStats(player)
    if not (coins and level) then return end
    coins.Value = coins.Value + GameMath.rewardForClick(level.Value)
    pushUpdate(player)
end)

buyEvent.OnServerEvent:Connect(function(player, upgradeId)
    if upgradeId ~= "multiplier" then return end

    local coins, level = getStats(player)
    if not (coins and level) then return end

    local cost = GameMath.upgradeCost(GameConfig.upgradeBaseCost, level.Value)
    if coins.Value >= cost then
        coins.Value = coins.Value - cost
        level.Value = level.Value + 1
    end

    pushUpdate(player)
end)
```

The click reports, the multiplier grows, the server owns the wallet.

</Variant>

<Variant name="collector">

```luau
local Players = game:GetService("Players")
local ReplicatedStorage = game:GetService("ReplicatedStorage")
local Shared = ReplicatedStorage.Shared
local GameConfig = require(Shared.GameConfig)
local GameMath = require(Shared.GameMath)

local buyEvent = ReplicatedStorage.RemoteEvents.BuyPet
local updateEvent = ReplicatedStorage.RemoteEvents.UpdateCoins

function getStats(player)
    local stats = player.leaderstats
    if not stats then return nil, nil end
    return stats[GameConfig.coinsKey], stats[GameConfig.levelKey]
end

function grant(player, amount)
    local coins, level = getStats(player)
    if not (coins and level) then return end
    coins.Value = coins.Value + amount
    updateEvent:FireClient(player, coins.Value, level.Value)
end

for _, pickup in ipairs(workspace.Pickups:GetChildren()) do
    pickup.Touched:Connect(function(other)
        local character = other.Parent
        local player = Players:GetPlayerFromCharacter(character)
        if not player then return end

        local coins, level = getStats(player)
        if not (coins and level) then return end

        pickup:Destroy()
        grant(player, GameMath.coinsPerPickup(level.Value))
    end)
end

buyEvent.OnServerEvent:Connect(function(player, petId)
    if petId ~= "slime" then return end

    local coins, level = getStats(player)
    if not (coins and level) then return end

    local cost = GameMath.upgradeCost(GameConfig.petBaseCost, level.Value)
    if coins.Value >= cost then
        coins.Value = coins.Value - cost
        level.Value = level.Value + 1
    end

    updateEvent:FireClient(player, coins.Value, level.Value)
end)
```

No remote needed for picking up a coin — the server owns physics too, so `Touched` fires where the truth lives. The pet purchase still goes through a remote.

## Step 4 — ClientUI

The whole client: build the UI, send actions, receive updates.

<Variant name="tycoon">

```luau

local ReplicatedStorage = game:GetService("ReplicatedStorage")

local screenGui = Instance.new("ScreenGui")
screenGui.Parent = player.PlayerGui

local coinsLabel = Instance.new("TextLabel")

coinsLabel.TextSize = 30
coinsLabel.BackgroundTransparency = 1
coinsLabel.Parent = screenGui

local coinButton = Instance.new("TextButton")

coinButton.AnchorPoint = Vector2.new(0.5, 0.5)
coinButton.Text = "Click!"
coinButton.TextSize = 26
coinButton.Parent = screenGui

local buyButton = Instance.new("TextButton")
buyButton.Size = UDim2.new(0, 240, 0, 60)
buyButton.Position = UDim2.new(0.5, 0, 0.7, 0)
buyButton.AnchorPoint = Vector2.new(0.5, 0.5)
buyButton.Text = "Buy upgrade (25 coins)"
buyButton.Parent = screenGui

coinButton.Activated:Connect(function()
    clickEvent:FireServer()
end)

buyButton.Activated:Connect(function()
    buyEvent:FireServer("multiplier")
end)

local function updateUI(coins, multiplier)
    coinsLabel.Text = "Coins: " .. coins
    buyButton.Text = "Buy upgrade (" .. 25 * (multiplier + 1) .. " coins)"
end

clickEvent.OnClientEvent:Connect(updateUI)
```

The button label even prices the next upgrade — but notice it's only *display*. The real price check lives on the server.

## Step 5 — polish

A game you just assembled still feels flat. Add feel with a tween — something that responds the instant the player does something.

<Variant name="tycoon">

A button that pops on every click is the difference between "a screen" and "a game":

```luau
coinButton.Activated:Connect(function()
    local TweenService = game:GetService("TweenService")
    local pop = TweenService:Create(coinButton, TweenInfo.new(0.1), {
        Size = UDim2.new(0, 260, 0, 150),
    })
    pop:Play()
    pop.Completed:Wait()
    TweenService:Create(coinButton, TweenInfo.new(0.1), {
        Size = UDim2.new(0, 240, 0, 140),
    }):Play()

    clickEvent:FireServer()
end)
```

</Variant>

<Variant name="collector">

A coin counter that pops every time a coin lands is the difference between "a HUD" and "a game":

```luau
local TweenService = game:GetService("TweenService")

local function popLabel()
    local pop = TweenService:Create(coinsLabel, TweenInfo.new(0.1), {
        TextSize = 36,
    })
    pop:Play()
    pop.Completed:Wait()
    TweenService:Create(coinsLabel, TweenInfo.new(0.1), {
        TextSize = 30,
    }):Play()
end

local function updateUI(coins, pets)
    coinsLabel.Text = "Coins: " .. coins
    buyButton.Text = "Buy a slime (" .. 30 * (pets + 1) .. " coins)"
    popLabel()
end

updateEvent.OnClientEvent:Connect(updateUI)
```

</Variant>

## Playtest and ship

Run this as your final test — the exact loop every game must pass: **earn, spend, verify, save, reload.**

<Variant name="tycoon">

1. **Earn:** click until you have 30 coins. Multiplier starts at x1.
2. **Spend:** buy the first upgrade. Coins drop by 25, multiplier becomes x2.
3. **Verify:** clicks now pay 2.
4. **Save:** stop the game.
5. **Reload:** play again. Coins *and* multiplier are back.

</Variant>

<Variant name="collector">

1. **Earn:** walk into coins until you have 40. Level starts at 1 pet.
2. **Spend:** buy a slime for 30. Level becomes 2.
3. **Verify:** each pickup now pays 2.
4. **Save:** stop the game.
5. **Reload:** play again. Coins *and* pets are back.

</Variant>

**The course is complete when step 5 shows your exact coins and level.**

- **Stretch goals** — in Coin Tycoon, an auto-income loop using `GameMath.rewardForClick`; in The Collector, different pets with different costs; in both, a "power" boost, a coin tween that flies to the corner, a third NumberValue for the leaderboard. Every one is a variation of the pipeline you've mastered.

- **Common mistakes, final edition** — skipping modules (the split *is* the debuggability); renaming a stat and missing the saves; testing saves without Studio access enabled; letting the client label decide anything (it's a mirror, not a bank).
